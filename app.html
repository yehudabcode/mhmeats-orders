// ══════════════════════════════════════════════════════════════════ // MH MEATS — Google Apps Script // Sheet name: "Orders" // Columns: Timestamp | Customer | CustomerName | Day | ProductKey | ProductName | Quantity | Status // ══════════════════════════════════════════════════════════════════ const SHEET_NAME = 'Orders'; // Column index constants (0-based) const COL = { TIMESTAMP: 0, CUSTOMER: 1, CUSTOMER_NAME:2, DAY: 3, PRODUCT_KEY: 4, PRODUCT_NAME: 5, QUANTITY: 6, STATUS: 7, }; const HEADERS = ['Timestamp','Customer','CustomerName','Day','ProductKey','ProductName','Quantity','Status']; // ── Ensure sheet exists with correct headers ────────────────────── function getSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); sheet.setFrozenRows(1); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold'); } return sheet; } // ══════════════════════════════════════════════════════════════════ // POST — receives order submissions from the customer portal // Payload: { customer, customerName, day, orders:{key:qty}, timestamp, notes? } // All submitted rows are written as 'active'. // Previous active rows are never touched — they accumulate until imported. // ══════════════════════════════════════════════════════════════════ function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const { customer, customerName, day, orders, timestamp, notes } = payload; if (!customer || !day || !orders) { return jsonResponse({ ok: false, error: 'Missing required fields' }); } const sheet = getSheet(); const ts = timestamp || new Date().toISOString(); // Write one row per product line — always active, never supersede anything Object.entries(orders).forEach(([productKey, quantity]) => { if (!quantity || parseInt(quantity) <= 0) return; sheet.appendRow([ ts, customer, customerName || customer, day, productKey, productKey, parseInt(quantity), 'active', ]); }); return jsonResponse({ ok: true }); } catch (err) { return jsonResponse({ ok: false, error: err.message }); } } // ══════════════════════════════════════════════════════════════════ // GET — serves data to the staff tool and portal (JSONP) // ══════════════════════════════════════════════════════════════════ function doGet(e) { const action = e.parameter.action || ''; const callback = e.parameter.callback || 'callback'; const customer = e.parameter.customer || ''; try { switch (action) { // ── Health check ──────────────────────────────────────────── case 'ping': return jsonpResponse(callback, { ok: true, ts: new Date().toISOString() }); // ── Staff tool: pull all active orders (marks them imported) ─ case 'getOrders': { const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const orders = []; const markRowNums = []; // 1-indexed sheet row numbers to flip to 'imported' for (let i = dataStart(data); i < data.length; i++) { const row = data[i]; if (row[COL.STATUS] !== 'active') continue; orders.push({ timestamp: row[COL.TIMESTAMP], customer: row[COL.CUSTOMER], customerName: row[COL.CUSTOMER_NAME], day: row[COL.DAY], productKey: row[COL.PRODUCT_KEY], productName: row[COL.PRODUCT_NAME], quantity: row[COL.QUANTITY], notes: '', }); markRowNums.push(i + 1); // sheet row number (1-indexed) } // Batch all status updates into a single write instead of one call per row. // Only touches the STATUS column, and only for rows that were pulled. if (markRowNums.length) { const statusRange = sheet.getRange(1, COL.STATUS + 1, sheet.getLastRow(), 1); const statusValues = statusRange.getValues(); markRowNums.forEach(rowNum => { statusValues[rowNum - 1][0] = 'imported'; }); statusRange.setValues(statusValues); } return jsonpResponse(callback, { ok: true, orders }); } // ── Portal: fetch customer's last submitted order ──────────── case 'getLastOrder': { if (!customer) return jsonpResponse(callback, { ok: false, orders: [] }); const sheet = getSheet(); const data = sheet.getDataRange().getValues(); // Find most recent timestamp for this customer let latestTs = ''; for (let i = dataStart(data); i < data.length; i++) { const row = data[i]; if (row[COL.CUSTOMER] !== customer) continue; const ts = row[COL.TIMESTAMP] || ''; if (ts > latestTs) latestTs = ts; } if (!latestTs) return jsonpResponse(callback, { ok: true, orders: [] }); // Return all rows from that timestamp const orders = []; for (let i = dataStart(data); i < data.length; i++) { const row = data[i]; if (row[COL.CUSTOMER] !== customer) continue; if (row[COL.TIMESTAMP] !== latestTs) continue; orders.push({ productKey: row[COL.PRODUCT_KEY], productName: row[COL.PRODUCT_NAME], quantity: row[COL.QUANTITY], day: row[COL.DAY], timestamp: row[COL.TIMESTAMP], }); } return jsonpResponse(callback, { ok: true, orders }); } // ── Portal: order history for a customer (last N days) ─────── case 'getHistory': { if (!customer) return jsonpResponse(callback, { ok: false, orders: [] }); const days = parseInt(e.parameter.days || '7'); const cutoff = new Date(); cutoff.setDate(cutoff.getDate() - days); const sheet = getSheet(); const data = sheet.getDataRange().getValues(); const orders = []; for (let i = dataStart(data); i < data.length; i++) { const row = data[i]; if (row[COL.CUSTOMER] !== customer) continue; const ts = row[COL.TIMESTAMP]; if (!ts || new Date(ts) < cutoff) continue; orders.push({ productKey: row[COL.PRODUCT_KEY], productName: row[COL.PRODUCT_NAME], quantity: row[COL.QUANTITY], day: row[COL.DAY], timestamp: row[COL.TIMESTAMP], notes: row[COL.NOTES] || '', }); } // Sort newest first orders.sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp)); return jsonpResponse(callback, { ok: true, orders }); } // ── Staff tool: reset imported rows back to active (emergency) ─ case 'resetImported': { const day = e.parameter.day || ''; const cust = e.parameter.customer || ''; const sheet = getSheet(); const data = sheet.getDataRange().getValues(); let count = 0; for (let i = dataStart(data); i < data.length; i++) { const row = data[i]; if (row[COL.STATUS] !== 'imported') continue; if (day && row[COL.DAY] !== day) continue; if (cust && row[COL.CUSTOMER] !== cust) continue; sheet.getRange(i + 1, COL.STATUS + 1).setValue('active'); count++; } return jsonpResponse(callback, { ok: true, reset: count }); } default: return jsonpResponse(callback, { ok: false, error: 'Unknown action: ' + action }); } } catch (err) { return jsonpResponse(callback, { ok: false, error: err.message }); } } // ══════════════════════════════════════════════════════════════════ // Helpers // ══════════════════════════════════════════════════════════════════ function jsonpResponse(callback, data) { return ContentService .createTextOutput(callback + '(' + JSON.stringify(data) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // Returns 1 to skip the header row, 0 if no header present. function dataStart(data) { if (!data || !data.length) return 1; const first = String(data[0][0] || '').toLowerCase().trim(); return (first === 'timestamp' || first === 'date') ? 1 : 0; }
